      <input type="number" id="amount" inputmode="decimal" step="0.01" placeholder="Например: 4500">
    </div>

    <div class="field">
      <label for="comment">Комментарий (назначение):</label>
      <textarea id="comment" placeholder="За что оплата / контрагент (необязательно)"></textarea>
    </div>

    <div class="field">
      <label>Документ или фото чека (при наличии):</label>
      <div class="file-box" onclick="document.getElementById('doc-file').click()">
        📎 Нажмите, чтобы прикрепить фото или PDF
        <div id="file-name" class="file-name">Файл не выбран</div>
      </div>
      <input type="file" id="doc-file" accept="image/*,.pdf,.doc,.docx,.xls,.xlsx" style="display: none;">
    </div>

  <script>
    const tg = window.Telegram.WebApp;
    tg.expand();

    // ВСТАВЬТЕ СЮДА ССЫЛКУ НА GOOGLE APPS SCRIPT ИЗ ШАГА 2
    const GAS_URL = "https://script.google.com/macros/s/AKfycbyvmK7KlUMjPTLbHsMcnSoM9NJb84kynsEE7WYzSMjgB5kPTFxbYjAiZ55FkItc9tjY/exec";

    // По умолчанию ставим сегодняшнюю дату
    const dateInput = document.getElementById('exp-date');
    dateInput.valueAsDate = new Date();

    let attachedFile = null;

    // Обработка выбора файла и перевод в Base64
    document.getElementById('doc-file').addEventListener('change', (e) => {
      const file = e.target.files[0];
      if (!file) {
        attachedFile = null;
        document.getElementById('file-name').innerText = "Файл не выбран";
        return;
      }
      if (file.size > 15 * 1024 * 1024) {
        tg.showAlert("Файл слишком большой (максимум 15 МБ).");
        e.target.value = "";
        return;
      }

      document.getElementById('file-name').innerText = `Выбран: ${file.name}`;
      const reader = new FileReader();
      reader.onload = () => {
        const base64 = reader.result.split(',')[1];
        attachedFile = {
          name: file.name,
          type: file.type,
          base64: base64
        };
      };
      reader.readAsDataURL(file);
    });

    // Загрузка точек и статей из Google Таблицы
    async function initApp() {
      try {
        const res = await fetch(GAS_URL);
        const data = await res.json();

        document.getElementById('location').innerHTML = data.locations
          .map(l => `<option value="${l}">${l}</option>`).join('');

        document.getElementById('category').innerHTML = data.categories
          .map(c => `<option value="${c}">${c}</option>`).join('');

        document.getElementById('loading').style.display = 'none';
        document.getElementById('app').style.display = 'block';

        tg.MainButton.text = "ОТПРАВИТЬ РАСХОД";
        tg.MainButton.show();
      } catch (err) {
        document.getElementById('loading').innerText = "Ошибка загрузки справочников. Проверьте ссылку GAS_URL.";
      }
    }

    // Отправка данных
    tg.MainButton.onClick(async () => {
      const rawDate = dateInput.value;
      const amount = parseFloat(document.getElementById('amount').value);

      if (!rawDate) {
        tg.showAlert("Укажите дату расхода!");
        return;
      }
      if (!amount || amount <= 0) {
        tg.showAlert("Введите корректную сумму расхода!");
        return;
      }

      // Форматируем дату из YYYY-MM-DD в DD.MM.YYYY
      const [y, m, d] = rawDate.split('-');
      const formattedDate = `${d}.${m}.${y}`;

      const user = tg.initDataUnsafe.user;
      const userId = user ? user.id : "";
      const userName = user
        ? (user.username ? `@${user.username}` : `${user.first_name || ""} ${user.last_name || ""}`.trim())
        : "Неизвестно";

      const payload = {
        date: formattedDate,
        location: document.getElementById('location').value,
        category: document.getElementById('category').value,
        amount: amount,
        comment: document.getElementById('comment').value.trim(),
        file: attachedFile,
        userId: userId,
        userName: userName
      };

      tg.MainButton.showProgress();
      tg.MainButton.disable();

      try {
        await fetch(GAS_URL, {
          method: 'POST',
          mode: 'no-cors',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload)
        });

        tg.MainButton.hideProgress();
        tg.close();
      } catch (err) {
        tg.MainButton.hideProgress();
        tg.MainButton.enable();
        tg.showAlert("Ошибка при отправке. Попробуйте ещё раз.");
      }
    });

    initApp();
  </script>
